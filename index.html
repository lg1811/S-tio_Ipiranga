async function carregarDadosIniciais() { try { const response = await fetch('/S-tio_Ipiranga/frutiferas.geojson'); const geojson = await response.json(); plantas = geojson.features.map(f => { const props = f.properties; return { id: props.fid || props.ID || props.id || Math.random(), nome: props.NOME_POPULAR || props.nome_popular || props.especie || props.nome || 'Sem identificação', nomeIdentificador: props.ID || props.fid || 'S/N', ultimaRega: props.ULTIMA_REGA || props.ultima_rega || null, ultimaPoda: props.ULTIMA_PODA || props.ultima_poda || null, ultimaColheita: props.ULTIMA_COLHEITA || props.ultima_colheita || null }; }); sessionStorage.setItem('plantas', JSON.stringify(plantas)); criarFiltros(); renderAll(); console.log(`✅ ${plantas.length} plantas carregadas do arquivo base`); } catch (err) { console.log('ℹ️ Arquivo base não encontrado, aguardando importação manual'); renderAll(); } } window.addEventListener('load', () => { loadConfig(); const saved = sessionStorage.getItem('plantas'); if (saved) { plantas = JSON.parse(saved); criarFiltros(); renderAll(); } else { carregarDadosIniciais(); } });
